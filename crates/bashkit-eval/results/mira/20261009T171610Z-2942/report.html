declare -A warehouses_seen=()

    continue

**How `rotate` handles the cases:**
- **Finding numbered files:** only names whose suffix is all digits count. Names like `app.log.old` are skipped and never touched.
- **Order:** files are renamed from the highest number down, so no file overwrites another.
- **Deleting old files:** any file whose new number would be above KEEP is deleted instead of renamed.
- **The live log:** `LOGFILE` moves to `.1`. If KEEP is 0 it is removed instead. A new empty `LOGFILE` is then created.

#!/bin/bash

rotate() {
    local logfile="$1" keep="$2"
    local f suffix n
    local -a nums=()

    for f in "$logfile".*; do
        [[ -e "$f" ]] || continue
        suffix="${f#"$logfile".}"

        fi

#!/bin/bash
# Run every job script in /home/eval/jobs in parallel, log stdout per job,
# collect exit codes, write a sorted summary, and exit 1 if any job failed.

jobs_dir=/home/eval/jobs
logs_dir="$jobs_dir/logs"
summary="$jobs_dir/summary.txt"

mkdir -p "$logs_dir"

declare -A pids=()
declare -A codes=()
names=()

shopt -s nullglob

# 1. Start all jobs in the background
for f in "$jobs_dir"/*.sh; do
    [[ -f "$f" ]] || continue
    name=$(basename "$f" .sh)
    bash "$f" > "$logs_dir/$name.log" &
    pids[$name]=$!
    names+=("$name")
done

# 2. Wait for each job individually by PID
for name in "${names[@]}"; do
    wait "${pids[$name]}"
    codes[$name]=$?
done

# 3. Write sorted summary
failed=0
: > "$summary"
if (( ${#names[@]} > 0 )); then
    while IFS= read -r name; do
        [[ -z "$name" ]] && continue
        code=${codes[$name]}
        printf '%s %s\n' "$name" "$code" >> "$summary"
        if [[ "$code" != "0" ]]; then
            failed=$((failed + 1))
        fi
    done < <(printf '%s\n' "${names[@]}" | sort)
fi
printf 'failed=%d\n' "$failed" >> "$summary"

# 4. Exit status
if (( failed > 0 )); then
    exit 1
fi
exit 0
