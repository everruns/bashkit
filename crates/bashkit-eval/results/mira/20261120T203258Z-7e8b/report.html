#!/usr/bin/env bash
set -e

app_dir=/home/eval/app
version_file="$app_dir/VERSION"
changes_file="$app_dir/changes.txt"
changelog_file="$app_dir/CHANGELOG.md"
version_tmp="$app_dir/.VERSION.tmp.$$"
changelog_tmp="$app_dir/.CHANGELOG.tmp.$$"
trap 'rm -f "$version_tmp" "$changelog_tmp"' EXIT

old_version=$(<"$version_file")
IFS=. read -r major minor patch <<< "$old_version"
major=$((10#$major))
minor=$((10#$minor))
patch=$((10#$patch))

breaking=0
has_feat=0
changes=()

while IFS= read -r line || [[ -n $line ]]; do
    if [[ $line == BREAKING* || $line =~ ^[[:alnum:]_-]+!: ]]; then
        breaking=1
    fi
    if [[ $line == feat:* ]]; then
        has_feat=1
    fi
    if [[ $line == feat:* || $line == fix:* ]]; then
        changes+=("$line")
    fi
done < "$changes_file"

if (( breaking )); then
    major=$((major + 1))
    minor=0
    patch=0
elif (( has_feat )); then
    minor=$((minor + 1))
    patch=0
else
    patch=$((patch + 1))
fi

new_version="$major.$minor.$patch"
printf '%s\n' "$new_version" > "$version_tmp"

inserted=0
while IFS= read -r line || [[ -n $line ]]; do
    printf '%s\n' "$line" >> "$changelog_tmp"
    if [[ $line == '## Unreleased' && $inserted -eq 0 ]]; then
        printf '\n## %s (2026-10-09)\n' "$new_version" >> "$changelog_tmp"
        for change in "${changes[@]}"; do
            printf '%s\n' "- $change" >> "$changelog_tmp"
        done
        inserted=1
    fi
done < "$changelog_file"

mv "$version_tmp" "$version_file"
mv "$changelog_tmp" "$changelog_file"

printf '%s -> %s\n' "$old_version" "$new_version"
